module m1 {
    var int1 int
    var bool1 bool
}

module m2 {
    var modvar module
}

main(
    m1.int1 = 3
    m1.bool1 = true

    % '==' and '!=' are only for numberical comparisons
    if (m1.bool1) {
        m1.int1 = 2
    } else if (bool1 && true) {
        m1.int1 = m1.int1 * 2
    } else {
        m1.int1 = -5
    }

    m2.modvar = m1
    m2.modvar.int1 = 5
)
